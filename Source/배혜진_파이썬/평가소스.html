def commadrop(row):
    row = row.copy()
    price = row['분양가격(제곱미터당 천원)']
    if pd.isna(price):
        row['평당분양가격'] = np.nan
    elif price.strip()=='' or price=='-' :
        row['평당분양가격'] = np.nan
    elif price.find(',') != -1:
        row['평당분양가격'] = float(price.replace(',', ''))
    else:
        row['평당분양가격'] = float(price)
        
    return row
